cmake_minimum_required(VERSION 3.16)
project(physics_engine C)

set(CMAKE_C_STANDARD 99)

include(FetchContent)
FetchContent_Declare(
    SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG        main
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(SDL3)

file(GLOB SOURCES "src/*.c")

add_executable(physics_engine ${SOURCES})
target_compile_options(physics_engine PRIVATE -O3 -Wextra)
target_include_directories(physics_engine PRIVATE include)
target_link_libraries(physics_engine PRIVATE SDL3::SDL3)

if(WIN32)
    add_custom_command(TARGET physics_engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:SDL3::SDL3>
            $<TARGET_FILE_DIR:physics_engine>
    )
endif()
